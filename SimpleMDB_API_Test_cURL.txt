# =============================================================================
# MOVIES (5 endpoints)
# =============================================================================

# 1. List Movies (GET /api/v1/movies?page=1&size=5)
curl -s "http://localhost:8080/api/v1/movies?page=1&size=5"

# 2. Create Movie (POST /api/v1/movies)
curl -s -X POST "http://localhost:8080/api/v1/movies" -H "Content-Type: application/json" -d "{\"title\":\"Test Movie\",\"year\":2024,\"genre\":\"Action\",\"rating\":7.5,\"description\":\"A test movie.\"}"

# 3. Read Movie (GET /api/v1/movies/1)
curl -s "http://localhost:8080/api/v1/movies/1"

# 4. Update Movie (PUT /api/v1/movies/1)
curl -s -X PUT "http://localhost:8080/api/v1/movies/1" -H "Content-Type: application/json" -d "{\"title\":\"The Godfather (Updated)\",\"year\":1972,\"genre\":\"Crime/Drama\",\"rating\":9.5,\"description\":\"Updated description.\"}"

# 5. Delete Movie (DELETE /api/v1/movies/50)
curl -s -X DELETE "http://localhost:8080/api/v1/movies/50"
